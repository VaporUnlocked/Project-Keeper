cmake_minimum_required(VERSION 3.20)
project(ProjectKeeper VERSION 0.6.0 LANGUAGES CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use a Windows x64 MSVC toolchain")
endif()
function(keeper_compile_settings target)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>)
endfunction()

add_library(ProjectKeeperCore STATIC src/vault.cpp src/github.cpp)
keeper_compile_settings(ProjectKeeperCore)
target_link_libraries(ProjectKeeperCore PUBLIC bcrypt advapi32 winhttp ole32 uuid)

function(keeper_executable target)
    add_executable(${target} WIN32 ${ARGN} src/app.rc)
    keeper_compile_settings(${target})
    target_link_options(${target} PRIVATE /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /MANIFEST:NO)
    target_link_libraries(${target} PRIVATE ProjectKeeperCore shell32)
endfunction()

set(desktop_sources src/main.cpp src/ui.cpp src/dialogs.cpp)
keeper_executable(ProjectKeeper ${desktop_sources})
target_link_libraries(ProjectKeeper PRIVATE comctl32 gdiplus dwmapi)

keeper_executable(ProjectKeeperMcp src/mcp_main.cpp src/mcp.cpp)
target_compile_definitions(ProjectKeeperMcp PRIVATE PK_MCP_HELPER)
